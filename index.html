function generateResolution() { const statusDiv = document.getElementById("status"); const companyName = document.getElementById("companyName").value; if (!companyName) { alert("يرجى إدخال اسم الشركة أولاً!"); return; } const partnerInputs = document.querySelectorAll(".partner-name"); const partnersData = []; partnerInputs.forEach(input => { const val = input.value ? input.value.trim() : ""; if (val !== "") { partnersData.push({ name: val }); } }); statusDiv.style.color = "#d69e2e"; statusDiv.innerText = "جاري إنشاء القرار..."; // استخدام طلب مباشر لمنع ذاكرة المتصفح المؤقتة (Cache Busting) fetch("template1.docx?t=" + Date.now()) .then(res => { if (!res.ok) throw new Error("تعذر الوصول لملف الوورد (كود الخطأ: " + res.status + ")"); return res.arrayBuffer(); }) .then(buffer => { const zip = new PizZip(buffer); const doc = new window.docxtemplater(zip, { paragraphLoop: true, linebreaks: true }); doc.render({ day: document.getElementById("day").value, gDate: document.getElementById("gDate").value, companyName: companyName, crNumber: document.getElementById("crNumber").value, location: document.getElementById("location").value, yearEnd: document.getElementById("yearEnd").value, yearEndShort: document.getElementById("yearEndShort").value, authorizedPerson: document.getElementById("authorizedPerson").value, partners: partnersData }); const out = doc.getZip().generate({ type: "blob" }); saveAs(out, "قرار_تعيين_محاسب_" + companyName + ".docx"); statusDiv.style.color = "#38a169"; statusDiv.innerText = "تم تنزيل القرار بنجاح!"; }) .catch(err => { console.error(err); statusDiv.style.color = "#e53e3e"; statusDiv.innerText = "تفاصيل الخطأ: " + err.message; }); }
