function generateResolution() { const statusDiv = document.getElementById("status"); const companyName = document.getElementById("companyName").value; if (!companyName) { alert("يرجى إدخال اسم الشركة أولاً!"); return; } const partnerInputs = document.querySelectorAll(".partner-name"); const partnersData = []; partnerInputs.forEach(input => { const val = input.value ? input.value.trim() : ""; if (val !== "") { // إضافة سطر جديد بعد كل اسم لضمان نزول الاسم التالي للسطر الأسفل partnersData.push({ name: val + "\n" }); } }); statusDiv.style.color = "#d69e2e"; statusDiv.innerText = "جاري إنشاء القرار..."; fetch("template1.docx") .then(res => res.arrayBuffer()) .then(buffer => { const zip = new PizZip(buffer); // تفعيل خيار linebreaks لتنفيذ السطر الجديد داخل الوورد const doc = new window.docxtemplater(zip, { paragraphLoop: true, linebreaks: true }); doc.render({ day: document.getElementById("day").value, gDate: document.getElementById("gDate").value, companyName: companyName, crNumber: document.getElementById("crNumber").value, location: document.getElementById("location").value, yearEnd: document.getElementById("yearEnd").value, yearEndShort: document.getElementById("yearEndShort").value, authorizedPerson: document.getElementById("authorizedPerson").value, partners: partnersData }); const out = doc.getZip().generate({ type: "blob" }); saveAs(out, "قرار_تعيين_محاسب_" + companyName + ".docx"); statusDiv.style.color = "#38a169"; statusDiv.innerText = "تم تنزيل القرار بنجاح!"; }) .catch(err => { statusDiv.style.color = "#e53e3e"; statusDiv.innerText = "حدث خطأ أثناء جلب ملف template1.docx"; }); }
